{{- /* llms/document.html — render the /llms.txt body. Context: dict "cfg" CONFIG "sources" MAP Returns: string Document shape follows the llms.txt proposal: one H1, an optional blockquote summary, optional prose, then H2 sections of link lists, with `## Optional` last. Empty sections are omitted rather than emitted as empty headings. Built as a slice of lines joined by explicit newlines. Every generated value passes through llms/text.html, and no value is emitted through an HTML output action, so the content reaches resources.FromString as plain text rather than HTML-escaped markup. */ -}} {{- $cfg := .cfg -}} {{- $sources := .sources -}} {{- $lines := slice -}} {{- $lines = $lines | append (printf "# %s" (partial "llms/text.html" (dict "text" $cfg.index.title "max" 0))) -}} {{- with $cfg.index.summary -}} {{- $lines = $lines | append "" -}} {{- $lines = $lines | append (printf "> %s" (partial "llms/text.html" (dict "text" . "max" 0))) -}} {{- end -}} {{- with $cfg.index.intro -}} {{- $lines = $lines | append "" -}} {{- range (split (strings.TrimSpace .) "\n") -}} {{- with strings.TrimSpace . -}} {{- $lines = $lines | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- range $cfg.sections -}} {{- $sec := . -}} {{- $pages := partial "llms/select.html" (dict "section" $sec "cfg" $cfg) -}} {{- if $pages -}} {{- $lines = $lines | append "" -}} {{- $lines = $lines | append (printf "## %s" (partial "llms/text.html" (dict "text" $sec.title "max" 0))) -}} {{- range $pages -}} {{- $lines = $lines | append (partial "llms/entry.html" (dict "page" . "sources" $sources)) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* --- optional: cross-links that are not page-level sections --- */ -}} {{- $optional := slice -}} {{- range $cfg.optional -}} {{- $url := .url -}} {{- /* Same-site URLs are resolved against the site base so a deployment subpath survives; external, protocol-relative, and scheme URLs are left exactly as written. */ -}} {{- if and (not (findRE `^[A-Za-z][A-Za-z0-9+.-]*:` $url)) (not (strings.HasPrefix $url "//")) -}} {{- $url = absURL (strings.TrimPrefix "/" $url) -}} {{- end -}} {{- $line := printf "- [%s](%s)" (partial "llms/text.html" (dict "text" .title "max" 0)) $url -}} {{- $optional = $optional | append $line -}} {{- end -}} {{- if $optional -}} {{- $lines = $lines | append "" -}} {{- $lines = $lines | append "## Optional" -}} {{- range $optional -}} {{- $lines = $lines | append . -}} {{- end -}} {{- end -}} {{- return printf "%s\n" (delimit $lines "\n") -}}
