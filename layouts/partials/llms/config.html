{{- /* llms/config.html — normalize data/llms.yaml into the configuration dict the llms partials read. Defaults and the data file are combined with `merge`, and only values that decide what is published get a guard: the opt-in flag, the index language, the source prefix, the language selector, and exclusion paths. `errorf` fails the build without halting the template, so a guard may keep what it reported. */ -}} {{- $raw := hugo.Data.llms -}} {{- $default_lang := hugo.Sites.Default.Language.Lang -}} {{- $langs := slice -}} {{- range hugo.Sites -}}{{- $langs = $langs | append .Language.Lang -}}{{- end -}} {{- $cfg := dict "enabled" false "index" (dict "enabled" true "language" $default_lang "title" "" "summary" "" "intro" "") "sources" (dict "enabled" true "prefix" "source" "all" true "langs" slice "exclude_sections" slice "exclude_pages" slice) "sections" slice "optional" slice -}} {{- /* Strict opt-in: only the YAML boolean true enables the feature. */ -}} {{- $enabled := false -}} {{- if $raw -}} {{- if not (reflect.IsMap $raw) -}} {{- errorf "llms: data/llms.yaml must be a map, got %T" $raw -}} {{- else if eq $raw.enabled true -}} {{- $enabled = true -}} {{- else if and $raw.enabled (ne $raw.enabled false) -}} {{- warnf "llms: data/llms.yaml `enabled` is %T %q, not the boolean true; treating the feature as disabled" $raw.enabled $raw.enabled -}} {{- end -}} {{- end -}} {{- /* A disabled site is never asked for a valid index language or section. */ -}} {{- if $enabled -}} {{- $cfg = merge $cfg (dict "enabled" true) -}} {{- /* `isset`, not `with`: a scalar `index: false` must fail the build, not leave the index switched on. */ -}} {{- if isset $raw "index" -}}{{- $cfg = merge $cfg (dict "index" (merge $cfg.index $raw.index)) -}}{{- end -}} {{- $index := merge $cfg.index (dict "enabled" (eq $cfg.index.enabled true)) -}} {{- if not (in $langs $index.language) -}} {{- errorf "llms: `index.language` is %q but this site defines only %s" $index.language (delimit $langs ", ") -}} {{- $index = merge $index (dict "language" $default_lang) -}} {{- end -}} {{- /* The indexed language's own title, never whichever site rendered first. */ -}} {{- if not $index.title -}} {{- range hugo.Sites -}}{{- if eq .Language.Lang $index.language -}}{{- $index = merge $index (dict "title" .Title) -}}{{- end -}}{{- end -}} {{- end -}} {{- $cfg = merge $cfg (dict "index" $index) -}} {{- if isset $raw "sources" -}}{{- $cfg = merge $cfg (dict "sources" (merge $cfg.sources $raw.sources)) -}}{{- end -}} {{- $sources := merge $cfg.sources (dict "enabled" (eq $cfg.sources.enabled true)) -}} {{- if ne (printf "%T" $sources.prefix) "string" -}} {{- errorf "llms: `sources.prefix` must be a string, got %T" $sources.prefix -}} {{- else -}} {{- /* Joined onto the publish root, so it must stay relative and inside it. */ -}} {{- $p := strings.TrimSuffix "/" $sources.prefix -}} {{- if or (eq $p "") (strings.Contains $p "://") (strings.HasPrefix $p "/") (strings.Contains $p "..") -}} {{- errorf "llms: `sources.prefix` must be a relative path such as \"source\"; got %q" $sources.prefix -}} {{- else -}} {{- $sources = merge $sources (dict "prefix" $p) -}} {{- end -}} {{- end -}} {{- /* `isset` keeps an explicit empty list meaning "no source languages". */ -}} {{- if isset $sources "languages" -}} {{- $v := $sources.languages -}} {{- if reflect.IsSlice $v -}} {{- $selected := slice -}} {{- range $v -}} {{- if not (in $langs .) -}} {{- errorf "llms: `sources.languages` lists %v, but this site defines only %s" . (delimit $langs ", ") -}} {{- else if not (in $selected .) -}} {{- $selected = $selected | append . -}} {{- end -}} {{- end -}} {{- $sources = merge $sources (dict "all" false "langs" $selected) -}} {{- else if eq $v "all" -}} {{- $sources = merge $sources (dict "all" true "langs" slice) -}} {{- else -}} {{- errorf "llms: `sources.languages` must be \"all\" or a list of language codes, got %T %v" $v $v -}} {{- end -}} {{- end -}} {{- /* Normalize exclusion paths so comparisons ignore stray slashes. */ -}} {{- range $key := slice "exclude_sections" "exclude_pages" -}} {{- $clean := slice -}} {{- range (index $sources $key) -}}{{- $clean = $clean | append (strings.Trim (printf "%v" .) "/") -}}{{- end -}} {{- $sources = merge $sources (dict $key $clean) -}} {{- end -}} {{- $cfg = merge $cfg (dict "sources" $sources) -}} {{- $sections := slice -}} {{- range $raw.sections -}} {{- $sections = $sections | append (merge (dict "title" "" "featured" slice "limit" 20) .) -}} {{- end -}} {{- $cfg = merge $cfg (dict "sections" $sections) -}} {{- if isset $raw "optional" -}}{{- $cfg = merge $cfg (dict "optional" $raw.optional) -}}{{- end -}} {{- end -}} {{- return $cfg -}}
