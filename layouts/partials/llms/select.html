{{- /* llms/select.html — resolve one configured /llms.txt section into an ordered list of pages. No publication happens here; this is pure selection. Context: dict "section" SECTION "cfg" CONFIG Returns: slice of Page Two selectors remain: `pages` (kept exactly as written) and `section` (published pages, newest first, with `featured` leading). Determinism matters because the same checkout must produce the same file on every build. Hugo's collection `sort` is stable, so ordering is expressed as two passes: path ascending first, then date descending. Equal dates therefore keep their path-ascending order instead of depending on map iteration. */ -}} {{- $sec := .section -}} {{- $title := $sec.title -}} {{- $pages := slice -}} {{- $seen := slice -}} {{- if eq $sec.kind "pages" -}} {{- range $sec.selector -}} {{- $ref := . -}} {{- $p := site.GetPage (printf "/%s" $ref) -}} {{- if not $p -}} {{- errorf "llms: section %q lists %q, which does not exist" $title $ref -}} {{- else if not (partial "llms/published.html" $p) -}} {{- warnf "llms: section %q lists %q, which is not published; omitting it" $title $ref -}} {{- else if not (in $seen $p.Path) -}} {{- $pages = $pages | append $p -}} {{- $seen = $seen | append $p.Path -}} {{- end -}} {{- end -}} {{- else if eq $sec.kind "section" -}} {{- $ref := strings.Trim $sec.selector "/" -}} {{- $node := site.GetPage (printf "/%s" $ref) -}} {{- if not $node -}} {{- errorf "llms: section %q selects section %q, which does not exist" $title $sec.selector -}} {{- else -}} {{- /* Filter first: unpublished pages must not consume the limit. */ -}} {{- $candidates := slice -}} {{- $eligible := slice -}} {{- range $node.RegularPagesRecursive -}} {{- if partial "llms/published.html" . -}} {{- $candidates = $candidates | append . -}} {{- $eligible = $eligible | append .Path -}} {{- end -}} {{- end -}} {{- /* Featured entries lead in the order written, and must belong to the selected section: a path from somewhere else is a data mistake, not a curatable edge case. */ -}} {{- range $sec.featured -}} {{- $p := site.GetPage (printf "/%s" .) -}} {{- if not $p -}} {{- errorf "llms: section %q features %q, which does not exist" $title . -}} {{- else if not (in $eligible $p.Path) -}} {{- errorf "llms: section %q features %q, which is not a published page in section %q" $title . $sec.selector -}} {{- else if not (in $seen $p.Path) -}} {{- $pages = $pages | append $p -}} {{- $seen = $seen | append $p.Path -}} {{- end -}} {{- end -}} {{- /* Featuring more pages than the limit keeps them all; the limit is a floor for featured entries and a cap for the chronological fill. */ -}} {{- $n := $sec.limit -}} {{- if gt (len $pages) $n -}}{{- $n = len $pages -}}{{- end -}} {{- $sorted := sort $candidates "Path" "asc" -}} {{- $sorted = sort $sorted "Date" "desc" -}} {{- range $sorted -}} {{- if lt (len $pages) $n -}} {{- if not (in $seen .Path) -}} {{- $pages = $pages | append . -}} {{- $seen = $seen | append .Path -}} {{- end -}} {{- end -}} {{- end -}} {{- $pages = first $n $pages -}} {{- end -}} {{- end -}} {{- return $pages -}}
