{{- /* llms/entry.html — one bullet in an /llms.txt section. Context: dict "page" PAGE "sources" MAP Returns: string, a single Markdown list line The link label points at the original source file when one was published, because that is the artifact this feature exists to expose. The rendered page is offered as a second link so a reader can see the HTML output and cite a stable URL; an HTML page linking to its source does not provide the reverse navigation for an agent starting from llms.txt. Pages with no source (generated taxonomy listings) link straight to their HTML page. Formatting is fixed rather than punctuation-aware: `- [Label](source): Description — [Rendered page](html)`, with no trailing periods. The source-less form drops the citation link. */ -}} {{- $page := .page -}} {{- $src := index .sources $page.Path -}} {{- /* `.Title` first, then `.LinkTitle`; an untitled homepage has neither, so fall back to the site title rather than rendering an empty label. An untitled regular page is an authoring mistake and gets no fallback. */ -}} {{- $title := $page.Title | default $page.LinkTitle -}} {{- if and (not $title) $page.IsHome -}} {{- $title = site.Title -}} {{- end -}} {{- $title = partial "llms/text.html" (dict "text" $title "max" 120) -}} {{- $url := $page.Permalink -}} {{- with $src -}}{{- $url = . -}}{{- end -}} {{- $desc := "" -}} {{- with $page.Description -}} {{- $desc = . -}} {{- else -}} {{- with $page.Summary -}}{{- $desc = . -}}{{- end -}} {{- end -}} {{- $line := printf "- [%s](%s)" $title $url -}} {{- with partial "llms/text.html" (dict "text" $desc "max" 200) -}} {{- $line = printf "%s: %s" $line . -}} {{- end -}} {{- if $src -}} {{- $line = printf "%s — [Rendered page](%s)" $line $page.Permalink -}} {{- end -}} {{- return $line -}}
