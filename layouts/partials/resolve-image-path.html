{{- /* resolve-image-path.html — Resolve a front matter image path Handles three formats: - Remote URL (https:// or protocol-relative //) — returned as-is - Absolute path (/img/photo.jpg) — returned as-is - Relative filename (photo.jpg) — resolved relative to the page's content directory using the provided .file context Set "absolute" to true when a consumer needs an absolute URL, such as OpenGraph or Twitter metadata. Local paths are passed through absURL without their leading slash so a baseURL subpath is retained. Usage: {{ $src = partial "resolve-image-path.html" (dict "src" $src "file" .File) }} */ -}} {{- $src := .src -}} {{- if and (not (hasPrefix $src "/")) (not (findRE "^(https?:)?//" $src)) -}} {{- with .file -}} {{- $src = path.Join "/" (path.Dir .Path) $src -}} {{- end -}} {{- end -}} {{- if and .absolute (not (urls.Parse $src).IsAbs) (not (hasPrefix $src "//")) -}} {{- $src = absURL (strings.TrimPrefix "/" $src) -}} {{- end -}} {{- return $src -}}
