{{- /* meta-social.html — OpenGraph and Twitter Card metadata. Hugo's embedded social templates do not resolve relative image filenames beside standalone content files. Resolve those paths through resolve-image-path.html before publishing social metadata. */ -}} {{- $page := . -}} {{/* --- Shared page metadata --- */}} {{- $title := $page.Title | default site.Title -}} {{- $description := $page.Description | default ($page.Summary | plainify) | default site.Params.description -}} {{- $locale := site.Language.Locale | default site.Language.Lang | default "en-us" -}} {{/* --- Resolve page images, or fall back to the site's default images --- */}} {{- $images := slice -}} {{- $imageFile := false -}} {{- range $page.Params.images -}} {{- with . -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- if and $images $page.File -}} {{- $imageFile = $page.File -}} {{- end -}} {{- if not $images -}} {{- range site.Params.images -}} {{- with . -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- end -}} {{/* Keep site-wide fallback images independent of the current page's .File. Otherwise a relative site image such as "pages/logo.jpg" would be incorrectly resolved inside the current post's content directory. */}} {{- $imageURLs := slice -}} {{- range $images -}} {{- $imageURL := partial "resolve-image-path.html" (dict "src" . "file" $imageFile "absolute" true ) -}} {{- $imageURLs = $imageURLs | append $imageURL -}} {{- end -}} {{/* --- OpenGraph page identity and description --- */}} {{- with $description }} {{- end }} {{/* --- OpenGraph article details --- */}} {{- if $page.IsPage }} {{- with $page.Section }} {{- end }} {{- if not $page.Date.IsZero }} {{- end }} {{/* Do not publish a modified time that predates the published time. */}} {{- if and (not $page.Lastmod.IsZero) (not ($page.Lastmod.Before $page.Date)) }} {{- end }} {{/* Article tags belong with the OpenGraph article metadata. */}} {{- range $page.Params.tags }} {{- end }} {{- end }} {{/* --- OpenGraph images --- */}} {{- range $imageURLs }} {{- end }} {{/* --- Twitter Card metadata --- */}} {{- if $imageURLs }} {{- with index $imageURLs 0 }} {{- end }} {{- end }} {{- with $description }} {{- end }}
